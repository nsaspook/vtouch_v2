144void eaDogM_WriteString(char *strPtr)
145{
146 uint8_t len = (uint8_t) strlen(strPtr);
147
148 E_TRACE;
149 wait_lcd_done();
150 wait_lcd_set();
151 CS_SetLow(); /* SPI select display */
152 if (len > (uint8_t) MAX_STRLEN) {
153 len = MAX_STRLEN;
154 }
155 memcpy(spi_link.txbuf, strPtr, len);
156#ifdef USE_LCD_DMA
157 DMAnCON0bits.EN = 0; /* disable DMA to change source count */
158 DMA1_SetSourceSize(len);
159 DMA1_SetDestinationSize(1);
160 DMAnCON0bits.EN = 1; /* enable DMA */
161#else
162 SPI1_ExchangeBlock(spi_link.txbuf, len);
163#endif
164 start_lcd(); // start DMA transfer
165}

365char * eaDogM_Scroll_String(char *strPtr)
366{
367 scroll_lock = true;
368 memcpy((void *) &Sstr[4][0], &Sstr[0][0], MAX_STRLEN); // move top line to old line buffer
369 memcpy((void *) &Sstr[0][0], &Sstr[1][0], MAX_STRLEN);
370 memcpy((void *) &Sstr[1][0], &Sstr[2][0], MAX_STRLEN);

373 scroll_line_pos = 4;

377/*
378 * mainline info scroll updater, should run a ~1 second ticks
379 */
380void eaDogM_Scroll_Task(void)
381{
382 if (!scroll_lock) {
383 return;
384 }
385
386 if (scroll_line_pos == 0) { // wait for last line display time period
387 scroll_lock = false;

395 scroll_lock = true;
