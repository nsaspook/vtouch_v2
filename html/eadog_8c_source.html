37/*
38 * Init the NHD-0420D3Z-NSW-BBW-V3 in 8-bit serial mode
39 * channel 1 DMA if configured
40 */
41bool init_display(void)
42{
43 spi_link.txbuf = lcd_dma_buf; // use MCC DMA buffer variable
44 memset(Sstr, ' ', NSB * LSB); // clear scroll buffer of junk
45
46#ifdef USE_LCD_DMA
47 DMA1_SetSCNTIInterruptHandler(clear_lcd_done);
48 DMA1_SetORIInterruptHandler(spi_byte);
49 DMA1_SetDMAPriority(2);
50#endif
51#ifdef NHD // uses MODE 3 on the Q84, https://newhavendisplay.com/content/specs/NHD-0420D3Z-NSW-BBW-V3.pdf
52#ifdef USEMCC_SPI
53#else
54 SPI1CON0bits.EN = 0;
55 // mode 3
56 SPI1CON1 = 0x20;
57 // SSET disabled; RXR suspended if the RxFIFO is full; TXR required for a transfer;
58 SPI1CON2 = 0x03;
59 // BAUD 0;
60 SPI1BAUD = 0x04; // 50kHz SCK
61 // CLKSEL MFINTOSC;
62 SPI1CLK = 0x02;
63 // BMODE every byte; LSBF MSb first; EN enabled; MST bus master;
64 SPI1CON0 = 0x83;
65 SPI1CON0bits.EN = 1;
66#endif
67 if (powerup) {
68 wdtdelay(LCD_PWR_DELAY); // > 400ms power up delay
69 }
70
71#ifdef USE_LCD_DMA
72 SPI1INTFbits.SPI1TXUIF = 0;
73 DMASELECT = 0; // use DMA1
74 DMAnCON0bits.EN = 0;
75 SPI1CON0bits.EN = 0;
76 SPI1CON2 = 0x02; // Received data is not stored in the FIFO
77 SPI1CON0bits.EN = 1;
78 DMAnCON1bits.DMODE = 0;
79 DMAnCON1bits.DSTP = 0;
80 DMAnCON1bits.SMODE = 1;
81 DMAnCON1bits.SMR = 0;
82 DMAnCON1bits.SSTP = 1;
83 DMAnSSA = (uint24_t) spi_link.txbuf;
84 DMAnCON0bits.DGO = 0;
85 DMAnCON0bits.EN = 1; /* enable DMA */
86 SPI1INTFbits.SPI1TXUIF = 1;
87 send_lcd_cmd_dma(LCD_CMD_BRI); // set back-light level
88 send_lcd_data_dma(NHD_BL_MED);
89 send_lcd_cmd_dma(LCD_CMD_CONT); // set display contrast
90 send_lcd_data_dma(NHD_CONT);
91 send_lcd_cmd_dma(LCD_CMD_ON); // display on
92 send_lcd_cmd_dma(LCD_CMD_CLR); // clear screen
93 wdtdelay(NHD_L_DELAY);
94 DMA1_StopTransfer();
95#else
96 send_lcd_cmd(LCD_CMD_BRI); // set back-light level
97 send_lcd_data(NHD_BL_LOW);
98 send_lcd_cmd(LCD_CMD_CONT); // set display contrast
99 send_lcd_data(NHD_CONT);
100 send_lcd_cmd(LCD_CMD_OFF); // display on
101 send_lcd_cmd(LCD_CMD_CLR); // clear screen
102 wdtdelay(NHD_L_DELAY);
103 DMA1_StopTransfer();
104#endif
105#endif
106 powerup = false; // only of the first display init call
107 return true;
108}

144void eaDogM_WriteString(char *strPtr)
145{
146 uint8_t len = (uint8_t) strlen(strPtr);
147
148 E_TRACE;
149 wait_lcd_done();
150 wait_lcd_set();
151 CS_SetLow(); /* SPI select display */
152 if (len > (uint8_t) MAX_STRLEN) {
153 len = MAX_STRLEN;
154 }
155 memcpy(spi_link.txbuf, strPtr, len);
156#ifdef USE_LCD_DMA
157 DMAnCON0bits.EN = 0; /* disable DMA to change source count */
158 DMA1_SetSourceSize(len);
159 DMA1_SetDestinationSize(1);
160 DMAnCON0bits.EN = 1; /* enable DMA */
161#else
162 SPI1_ExchangeBlock(spi_link.txbuf, len);
163#endif
164 start_lcd(); // start DMA transfer
165}

213{

316{

365char * eaDogM_Scroll_String(char *strPtr)
366{
367 scroll_lock = true;

369 memcpy((void *) &Sstr[0][0], &Sstr[1][0], MAX_STRLEN);
370 memcpy((void *) &Sstr[1][0], &Sstr[2][0], MAX_STRLEN);

373 scroll_line_pos = 4;

377/*
378 * mainline info scroll updater, should run a ~1 second ticks
379 */
380void eaDogM_Scroll_Task(void)
381{
382 if (!scroll_lock) {
383 return;
384 }
385
386 if (scroll_line_pos == 0) { // wait for last line display time period
387 scroll_lock = false;
388 scroll_line_pos = 4;
389 return;
390 }
391
392 scroll_lock = false;
393 --scroll_line_pos;
394 eaDogM_WriteStringAtPos(scroll_line_pos, 0, &Sstr[scroll_line_pos][0]);
395 scroll_lock = true;
396
