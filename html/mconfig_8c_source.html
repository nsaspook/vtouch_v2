81uint8_t update_lcd(uint8_t vterm)
82{
83 vterm = vterm & 0x03; // mask to 2 bits for buffer array

85 D.lcd[vterm][0][MAX_LINE] = 0;
86 D.lcd[vterm][1][MAX_LINE] = 0;

89 wait_lcd_done();
90 eaDogM_WriteStringAtPos(0, 0, D.lcd[vterm][0]);

92 eaDogM_WriteStringAtPos(1, 0, D.lcd[vterm][1]);

94 eaDogM_WriteStringAtPos(2, 0, D.lcd[vterm][2]);
95 wait_lcd_done();
96 eaDogM_WriteStringAtPos(3, 0, D.lcd[vterm][3]);
