87void UART2_Initialize(void)
88{
89 // Disable interrupts before changing states
90 PIE8bits.U2RXIE = 0;
91 UART2_SetRxInterruptHandler(UART2_Receive_ISR);
92 PIE8bits.U2TXIE = 0;
93 UART2_SetTxInterruptHandler(UART2_Transmit_ISR);

98 U2P1L = 0x00;

101 U2P1H = 0x00;
102
103 // P2L 0;
104 U2P2L = 0x00;
105

107 U2P2H = 0x00;
108

110 U2P3L = 0x00;
111

113 U2P3H = 0x00;
114
115 // BRGS high speed; MODE Asynchronous 8-bit mode; RXEN enabled; TXEN enabled; ABDEN disabled;
116 U2CON0 = 0xB0;
117
118 // RXBIMD Set RXBKIF on rising RX input; BRKOVR disabled; WUE disabled; SENDB disabled; ON enabled;
119 U2CON1 = 0x80;
120
121 // TXPOL not inverted; FLO off; C0EN Checksum Mode 0; RXPOL not inverted; RUNOVF RX input shifter stops all activity; STP Transmit 1Stop bit, receiver verifies first Stop bit;
122 U2CON2 = 0x00;
123
124 // BRGL 130;
125 U2BRGL = 0x82;
126
127 // BRGH 6;
128 U2BRGH = 0x06;
129
130 // STPMD in middle of first Stop bit; TXWRE No error;
131 U2FIFO = 0x00;
132
133 // ABDIF Auto-baud not enabled or not complete; WUIF WUE not enabled by software; ABDIE disabled;
134 U2UIR = 0x00;
135
136 // ABDOVF Not overflowed; TXCIF 0; RXBKIF No Break detected; RXFOIF not overflowed; CERIF No Checksum error;
137 U2ERRIR = 0x00;
138
139 // TXCIE disabled; FERIE disabled; TXMTIE disabled; ABDOVE disabled; CERIE disabled; RXFOIE disabled; PERIE disabled; RXBKIE disabled;
140 U2ERRIE = 0x00;

144 UART2_SetOverrunErrorHandler(UART2_DefaultOverrunErrorHandler);

181uint8_t UART2_Read(void)
182{
183 uint8_t readValue = 0;
184
185 while (0 == uart2RxCount) {
186 }
187
188 uart2RxLastError = uart2RxStatusBuffer[uart2RxTail];
189
190 readValue = uart2RxBuffer[uart2RxTail++];
191 if (sizeof(uart2RxBuffer) <= uart2RxTail) {
192 uart2RxTail = 0;
193 }
194 PIE8bits.U2RXIE = 0;
195 uart2RxCount--;
196 PIE8bits.U2RXIE = 1;
197
198 return readValue;
199}

201void UART2_Write(uint8_t txData)
202{
203 while (0 == uart2TxBufferRemaining) {
204 }
205
206 if (0 == PIE8bits.U2TXIE) {
207 U2TXB = txData;
208 } else {
209 PIE8bits.U2TXIE = 0;
210 uart2TxBuffer[uart2TxHead++] = txData;
211 if (sizeof(uart2TxBuffer) <= uart2TxHead) {
212 uart2TxHead = 0;
213 }
214 uart2TxBufferRemaining--;
215 }
216 PIE8bits.U2TXIE = 1;
217}

234{
235 // use this default transmit interrupt handler code
236 if (sizeof(uart2TxBuffer) > uart2TxBufferRemaining) {
237 U2TXB = uart2TxBuffer[uart2TxTail++];
238 if (sizeof(uart2TxBuffer) <= uart2TxTail) {

249void UART2_Receive_ISR(void)
250{
251 // use this default receive interrupt handler code
252 uart2RxStatusBuffer[uart2RxHead].status = 0;
253
254 if (U2ERRIRbits.FERIF) {
255 uart2RxStatusBuffer[uart2RxHead].ferr = 1;
256 UART2_FramingErrorHandler();
257 }
258
259 if (U2ERRIRbits.RXFOIF) {
260 uart2RxStatusBuffer[uart2RxHead].oerr = 1;
261 UART2_OverrunErrorHandler();
262 }
263
264 if (uart2RxStatusBuffer[uart2RxHead].status) {
265 UART2_ErrorHandler();
266 } else {
267 UART2_RxDataHandler();
268 }
269
270 // or set custom function using UART2_SetRxInterruptHandler()
271}

273void UART2_RxDataHandler(void)
274{
275 // use this default receive interrupt handler code
276 uart2RxBuffer[uart2RxHead++] = U2RXB;
277 if (sizeof(uart2RxBuffer) <= uart2RxHead) {
278 uart2RxHead = 0;
279 }
280 uart2RxCount++;
281}

323void UART2_put_buffer(uint8_t bufData)
324{
325 INTERRUPT_GlobalInterruptHighDisable();
326 INTERRUPT_GlobalInterruptLowDisable();
327 PIE8bits.U2RXIE = 0;
328 uart2RxBuffer[uart2RxHead++] = bufData;
329 if (sizeof(uart2RxBuffer) <= uart2RxHead) {
330 uart2RxHead = 0;
331 }
332
333 uart2RxCount++;
334 PIE8bits.U2RXIE = 1;
335 INTERRUPT_GlobalInterruptLowEnable();
336 INTERRUPT_GlobalInterruptHighEnable();
337}
